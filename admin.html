function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var sheetName = data.type === 'registration' ? 'Burtgel' : 'Sudalgaa';
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  if (sheet.getLastRow() === 0) {
    if (data.type === 'registration') {
      sheet.appendRow(['Огноо','Байгууллага','Овог','Нэр','Утас','Оролцоо','Имэйл','ID']);
    } else {
      sheet.appendRow(['Огноо','Байршил','Нэр','Утас','Үнэлгээ','Эрсдэл','ШинэАрга','Тоглоом','Шийдвэр','Санал','ID']);
    }
  }
  if (data.type === 'registration') {
    sheet.appendRow([data.submittedAt, data.org, data.lastName, data.firstName, data.phone, data.role, data.email, data.id]);
  } else {
    sheet.appendRow([data.submittedAt, data.session_location, data.participant_name, data.participant_phone, data.q1_overall, data.q2_risk, data.q3_communication, data.q4_game, data.q5_action, data.q6_feedback, data.id]);
  }
  return ContentService.createTextOutput("OK");
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var regSheet = ss.getSheetByName('Burtgel');
  var srvSheet = ss.getSheetByName('Sudalgaa');
  var regs = regSheet && regSheet.getLastRow() > 1 ? regSheet.getDataRange().getValues().slice(1) : [];
  var srvs = srvSheet && srvSheet.getLastRow() > 1 ? srvSheet.getDataRange().getValues().slice(1) : [];
  return ContentService.createTextOutput(JSON.stringify({ registrations: regs, surveys: srvs }))
    .setMimeType(ContentService.MimeType.JSON);
}
